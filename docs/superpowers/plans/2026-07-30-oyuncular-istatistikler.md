# Oyuncular ve İstatistikler Sayfaları Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add public `/players`, `/players/:id`, `/statistics` pages plus their backing API endpoints and admin player-management extensions to the existing volleyball tournament app, computing all statistics dynamically from existing `Week`/`Team`/`TeamPlayer`/`Match`/`WeeklyMvp` data.

**Architecture:** Layered exactly like the existing `matchScoring.ts` → `standings.ts` → `weekView.ts` stack: pure calculation functions (no Prisma) in `src/lib/*.ts`, thin Prisma-querying "service" files that feed raw data into the pure functions, and thin Next.js Route Handlers / Server Components that call the services. No new tables — one migration adds columns + indexes to the existing `Player`/`WeekAttendee`/`TeamPlayer`/`WeeklyMvp`/`Match` tables.

**Tech Stack:** Next.js 14.2 App Router, TypeScript strict, Prisma 5.18 (Postgres), Tailwind 3.4 utility classes, Vitest 2.0.

Reference spec: `docs/superpowers/specs/2026-07-30-oyuncular-istatistikler-design.md`

## Global Constraints

- No new npm dependencies — everything is built with the existing stack (Next.js, Prisma, Tailwind, Vitest). Do not add a UI kit, state library, or Prisma alternative.
- All database access goes through the singleton Prisma client at `src/lib/db.ts` (`import { prisma } from '@/lib/db'`) — never instantiate a new `PrismaClient`.
- All new/changed public API routes live outside `/api/admin` and require no auth code (middleware does not protect them). All admin routes stay under `/api/admin` (already protected by `src/middleware.ts` — do not add auth checks inside route handlers).
- Turkish UI copy throughout, matching existing tone (e.g. "Oyuncular", "Geçmiş Haftalar", "Seviye 1-10 arasında olmalı").
- Tailwind utility classes only, inline in JSX — no CSS Modules, no styled-components, no new global CSS.
- `Player.notes` must never appear in any public-facing response or type (`/api/players`, `/api/players/:id`, and the pages that call them). It is admin-only.
- `Player.position` values are restricted to the `PLAYER_POSITIONS` constant — never accept or render an arbitrary string as a position.
- No `Season` model/field, no persistent cross-week `Team` identity, no cross-week team leaderboard — these were explicitly ruled out in the design spec.
- "Oynanmış maç" is defined solely by `isCompletedMatch()` (Task 4) — no other file re-implements `sets.length >= 2` after this plan is complete.
- Do not modify `src/lib/matchScoring.ts`'s existing exports' behavior, `src/lib/standings.ts`, `src/lib/teamBalancer.ts`, `src/middleware.ts`, or any existing admin week/team/match/mvp route — this plan only adds new files and additively extends `Player`-related files.
- Every task ends with: `npx tsc --noEmit`, `npm run lint`, the relevant Vitest run, and `npm run build` all passing (or documented as pre-existing failures per Task 1), followed by one commit for that task.

```ts
export const PLAYER_POSITIONS = ['Pasör', 'Çaprazcı', 'Smaçör', 'Orta Oyuncu', 'Libero'] as const;

export type PlayerPosition = (typeof PLAYER_POSITIONS)[number];

export function isValidPlayerPosition(value: unknown): value is PlayerPosition {
  return typeof value === 'string' && (PLAYER_POSITIONS as readonly string[]).includes(value);
}

export function isValidAvatarUrl(value: unknown): value is string {
  if (typeof value !== 'string' || value.trim().length === 0) return false;
  try {
    const url = new URL(value);
    return url.protocol === 'http:' || url.protocol === 'https:';
  } catch {
    return false;
  }
}
```

```ts
export interface MatchLike {
  sets: SetScore[];
  status?: 'SCHEDULED' | 'COMPLETED' | 'CANCELLED'; // Match.status does not exist in the DB yet; when it's added, wire it through here and every caller of isCompletedMatch benefits automatically.

export function isCompletedMatch(match: MatchLike): boolean {
  if (match.status) {
    return match.status === 'COMPLETED';
  }
  if (match.sets.length < 2) return false;
  try {
    computeMatchResult(match.sets);
    return true;
  } catch {
    return false;
  }
}
```

```ts
  teamAId: string;
  teamBId: string;

export interface PlayerMatchRecord {
  matchId: string;
  weekId: string;
  weekDate: Date;
  matchCreatedAt: Date;
  teamId: string;
  opponentTeamId: string;
  won: boolean;
  setsWonByPlayer: number;
  setsLostByPlayer: number;
}

export interface PlayerStatsResult {
  matchesPlayed: number;
  wins: number;
  losses: number;
  winRate: number;
  setsWon: number;
  setsLost: number;
  setDiff: number;
  weeksPlayed: number;
  currentWinStreak: number;
  longestWinStreak: number;
  last10: Array<'W' | 'L'>;
  records: PlayerMatchRecord[];
  integrityWarnings: string[];
}

export function countDistinctWeeks(weekIds: string[]): number {
  return new Set(weekIds).size;
}

export function computePlayerStats(assignments: PlayerMatchAssignment[]): PlayerStatsResult {
  const byMatch = new Map<string, PlayerMatchAssignment[]>();
  for (const assignment of assignments) {
    const list = byMatch.get(assignment.matchId) ?? [];
    list.push(assignment);
    byMatch.set(assignment.matchId, list);
  }

  const integrityWarnings: string[] = [];
  const records: PlayerMatchRecord[] = [];

  for (const [matchId, group] of byMatch) {
    if (group.length > 1) {
      integrityWarnings.push(
        `${matchId}: oyuncu maçın her iki takımında da kayıtlı, istatistiklere dahil edilmedi`
      );
      continue;
    }

    const assignment = group[0];
    const result = computeMatchResult(assignment.sets);
    const playerIsTeamA = assignment.teamId === assignment.teamAId;
    const won = playerIsTeamA ? result.winner === 'A' : result.winner === 'B';
    const setsWonByPlayer = playerIsTeamA ? result.setsWonA : result.setsWonB;
    const setsLostByPlayer = playerIsTeamA ? result.setsWonB : result.setsWonA;
    const opponentTeamId = playerIsTeamA ? assignment.teamBId : assignment.teamAId;

    records.push({
      matchId,
      weekId: assignment.weekId,
      weekDate: assignment.weekDate,
      matchCreatedAt: assignment.matchCreatedAt,
      teamId: assignment.teamId,
      opponentTeamId,
      won,
      setsWonByPlayer,
      setsLostByPlayer,
    });
  }

  records.sort((a, b) => {
    const dateDiff = a.weekDate.getTime() - b.weekDate.getTime();
    if (dateDiff !== 0) return dateDiff;
    const createdDiff = a.matchCreatedAt.getTime() - b.matchCreatedAt.getTime();
    if (createdDiff !== 0) return createdDiff;
    return a.matchId.localeCompare(b.matchId);
  });

  const matchesPlayed = records.length;
  const wins = records.filter((r) => r.won).length;
  const losses = matchesPlayed - wins;
  const winRate = matchesPlayed > 0 ? (wins / matchesPlayed) * 100 : 0;
  const setsWon = records.reduce((sum, r) => sum + r.setsWonByPlayer, 0);
  const setsLost = records.reduce((sum, r) => sum + r.setsLostByPlayer, 0);
  const weeksPlayed = countDistinctWeeks(records.map((r) => r.weekId));

  let longestWinStreak = 0;
  let running = 0;
  for (const r of records) {
    running = r.won ? running + 1 : 0;
    if (running > longestWinStreak) longestWinStreak = running;
  }

  let currentWinStreak = 0;
  for (let i = records.length - 1; i >= 0; i--) {
    if (records[i].won) currentWinStreak++;
    else break;
  }

  const last10: Array<'W' | 'L'> = records.slice(-10).map((r) => (r.won ? 'W' : 'L'));

  return {
    matchesPlayed,
    wins,
    losses,
    winRate,
    setsWon,
    setsLost,
    setDiff: setsWon - setsLost,
    weeksPlayed,
    currentWinStreak,
    longestWinStreak,
    last10,
    records,
    integrityWarnings,
  };
}
```

```ts
import { computeMatchResult, type SetScore } from './matchScoring';

export interface MatchRecordInput {
  matchId: string;
  weekId: string;
  weekDate: Date;
  teamAName: string;
  teamBName: string;
  sets: SetScore[];
}

export interface MatchRecordSummary {
  matchId: string;
  weekId: string;
  weekDate: Date;
  teamAName: string;
  teamBName: string;
  sets: SetScore[];
  winnerName: string;
  loserName: string;
  totalPointDiff: number;
  totalPoints: number;
}

function summarize(match: MatchRecordInput): MatchRecordSummary {
  const result = computeMatchResult(match.sets);
  const totalPointDiff = match.sets.reduce((sum, [a, b]) => sum + Math.abs(a - b), 0);
  const totalPoints = match.sets.reduce((sum, [a, b]) => sum + a + b, 0);
  const winnerName = result.winner === 'A' ? match.teamAName : match.teamBName;
  const loserName = result.winner === 'A' ? match.teamBName : match.teamAName;
  return {
    matchId: match.matchId,
    weekId: match.weekId,
    weekDate: match.weekDate,
    teamAName: match.teamAName,
    teamBName: match.teamBName,
    sets: match.sets,
    winnerName,
    loserName,
    totalPointDiff,
    totalPoints,
  };
}

export function findBiggestBlowout(matches: MatchRecordInput[]): MatchRecordSummary | null {
  if (matches.length === 0) return null;
  return matches.map(summarize).sort((a, b) => b.totalPointDiff - a.totalPointDiff)[0];
}

export function findClosestMatch(matches: MatchRecordInput[]): MatchRecordSummary | null {
  if (matches.length === 0) return null;
  return matches.map(summarize).sort((a, b) => a.totalPointDiff - b.totalPointDiff)[0];
}

export function findHighestScoringMatch(matches: MatchRecordInput[]): MatchRecordSummary | null {
  if (matches.length === 0) return null;
  return matches.map(summarize).sort((a, b) => b.totalPoints - a.totalPoints)[0];
}
```

```ts
import { prisma } from '@/lib/db';
import { isCompletedMatch, type SetScore } from '@/lib/matchScoring';
import type { PlayerMatchAssignment } from '@/lib/playerStats';

export async function getPlayerMatchAssignments(
  weekIds?: string[]
): Promise<Map<string, PlayerMatchAssignment[]>> {
  const matches = await prisma.match.findMany({
    where: weekIds ? { weekId: { in: weekIds } } : undefined,
    include: {
      week: true,
      teamA: { include: { players: { select: { playerId: true } } } },
      teamB: { include: { players: { select: { playerId: true } } } },
    },
  });

  const byPlayer = new Map<string, PlayerMatchAssignment[]>();

  for (const match of matches) {
    const sets = match.sets as SetScore[];
    if (!isCompletedMatch({ sets })) continue;

    const base = {
      matchId: match.id,
      weekId: match.weekId,
      weekDate: match.week.date,
      matchCreatedAt: match.createdAt,
      teamAId: match.teamAId,
      teamBId: match.teamBId,
      sets,
    };

    for (const { playerId } of match.teamA.players) {
      const list = byPlayer.get(playerId) ?? [];
      list.push({ ...base, teamId: match.teamAId });
      byPlayer.set(playerId, list);
    }
    for (const { playerId } of match.teamB.players) {
      const list = byPlayer.get(playerId) ?? [];
      list.push({ ...base, teamId: match.teamBId });
      byPlayer.set(playerId, list);
    }
  }

  return byPlayer;
}
```

```ts
  const minMatches = filters.minMatches ?? DEFAULT_MIN_MATCHES_FOR_WIN_RATE;
  const limit = Math.min(filters.limit ?? DEFAULT_LIMIT, MAX_LIMIT);
  const offset = filters.offset ?? 0;

  const [players, assignmentsByPlayer, mvpGroups] = await Promise.all([
    prisma.player.findMany({ where: { isActive: true } }),
    getPlayerMatchAssignments(),
    prisma.weeklyMvp.groupBy({ by: ['playerId'], _count: { playerId: true } }),
  ]);

  const mvpCountMap = new Map(mvpGroups.map((g) => [g.playerId, g._count.playerId]));

  let summaries: PlayerPublicSummary[] = players.map((player) => {
    const stats = computePlayerStats(assignmentsByPlayer.get(player.id) ?? []);
    return {
      id: player.id,
      name: player.name,
      position: player.position,
      level: player.level,
      avatarUrl: player.avatarUrl,
      isActive: player.isActive,
      weeksPlayed: stats.weeksPlayed,
      matchesPlayed: stats.matchesPlayed,
      wins: stats.wins,
      losses: stats.losses,
      winRate: stats.winRate,
      mvpCount: mvpCountMap.get(player.id) ?? 0,
    };
  });

  if (filters.search) {
    const term = filters.search.trim().toLowerCase();
    summaries = summaries.filter((p) => p.name.toLowerCase().includes(term));
  }
  if (filters.position) {
    summaries = summaries.filter((p) => p.position === filters.position);
  }
  if (typeof filters.skillLevel === 'number') {
    summaries = summaries.filter((p) => p.level === filters.skillLevel);
  }

  const sortBy = filters.sortBy ?? 'name';
  if (sortBy === 'winRate') {
    summaries = summaries.filter((p) => p.matchesPlayed >= minMatches);
  }
  const order = filters.order ?? (sortBy === 'name' ? 'asc' : 'desc');
  const direction = order === 'asc' ? 1 : -1;

  summaries.sort((a, b) => {
    switch (sortBy) {
      case 'matches':
        return (a.matchesPlayed - b.matchesPlayed) * direction;
      case 'wins':
        return (a.wins - b.wins) * direction;
      case 'winRate':
        return (a.winRate - b.winRate) * direction;
      case 'mvps':
        return (a.mvpCount - b.mvpCount) * direction;
      case 'name':
      default:
        return a.name.localeCompare(b.name) * direction;
    }
  });

  const total = summaries.length;
  const page = summaries.slice(offset, offset + limit);

  return { players: page, total, limit, offset, minMatchesForWinRate: minMatches };
}
```

```ts
export type StatisticsScope =
  | { type: 'all' }
  | { type: 'last10' }
  | { type: 'week'; weekId: string };

export function parseStatisticsScope(
  params: URLSearchParams
): { scope: StatisticsScope } | { error: string } {
  const scope = params.get('scope') ?? 'all';
  if (scope === 'all') return { scope: { type: 'all' } };
  if (scope === 'last10') return { scope: { type: 'last10' } };
  if (scope === 'week') {
    const weekId = params.get('weekId');
    if (!weekId) return { error: 'week kapsamı için weekId gerekli' };
    return { scope: { type: 'week', weekId } };
  }
  return { error: 'Geçersiz scope değeri' };
}

async function getLastNPlayedWeekIds(n: number): Promise<string[]> {

    .filter((w) => w.matches.some((m) => isCompletedMatch({ sets: m.sets as SetScore[] })))
    .slice(0, n)
    .map((w) => w.id);
}

  totalWeeks: number;
  totalMatches: number;
  totalSets: number;
  averageWeeklyAttendance: number;
  mostAttendedWeek: { weekId: string; weekDate: string; attendeeCount: number } | null;
}

export async function getOverview(scope: StatisticsScope): Promise<OverviewStats> {
  const weekIds = await resolveScopeWeekIds(scope);

  const [weeks, totalPlayers] = await Promise.all([
    prisma.week.findMany({
      where: weekIds ? { id: { in: weekIds } } : undefined,
      include: { matches: true, attendees: true },
    }),
    prisma.player.count({ where: { isActive: true } }),
  ]);

  const completedMatches = weeks.flatMap((w) =>
    w.matches.filter((m) => isCompletedMatch({ sets: m.sets as SetScore[] }))
  );
  const totalSets = completedMatches.reduce((sum, m) => sum + (m.sets as SetScore[]).length, 0);

  const attendanceCounts = weeks.map((w) => ({
    weekId: w.id,
    weekDate: w.date.toISOString(),
    attendeeCount: w.attendees.length,
  }));
  const averageWeeklyAttendance =
    weeks.length > 0
      ? attendanceCounts.reduce((sum, a) => sum + a.attendeeCount, 0) / weeks.length
      : 0;
  const mostAttendedWeek =
    attendanceCounts.length > 0
      ? attendanceCounts.reduce((max, a) => (a.attendeeCount > max.attendeeCount ? a : max))
      : null;

  return {
    totalPlayers,
    totalWeeks: weeks.length,
    totalMatches: completedMatches.length,
    totalSets,
    averageWeeklyAttendance,
    mostAttendedWeek,
  };
}

export interface LeaderboardEntry {
  playerId: string;
  name: string;
  value: number;
}

export interface PlayerLeaderboards {
  minMatchesForWinRate: number;
  mostMatches: LeaderboardEntry[];
  mostWins: LeaderboardEntry[];
  highestWinRate: LeaderboardEntry[];
  mostMvps: LeaderboardEntry[];
  mostWeeksPlayed: LeaderboardEntry[];
  mostSetsWon: LeaderboardEntry[];
  bestSetDiff: LeaderboardEntry[];
}

const LEADERBOARD_SIZE = 10;
const DEFAULT_MIN_MATCHES_FOR_WIN_RATE = 3;

export async function getPlayerLeaderboards(scope: StatisticsScope): Promise<PlayerLeaderboards> {
  const weekIds = await resolveScopeWeekIds(scope);

  const [players, assignmentsByPlayer, mvpGroups] = await Promise.all([
    prisma.player.findMany({ where: { isActive: true } }),
    getPlayerMatchAssignments(weekIds),
    prisma.weeklyMvp.groupBy({
      by: ['playerId'],
      _count: { playerId: true },
      where: weekIds ? { weekId: { in: weekIds } } : undefined,
    }),
  ]);
  const mvpCountMap = new Map(mvpGroups.map((g) => [g.playerId, g._count.playerId]));

  const rows = players.map((player) => {
    const stats = computePlayerStats(assignmentsByPlayer.get(player.id) ?? []);
    return {
      playerId: player.id,
      name: player.name,
      matchesPlayed: stats.matchesPlayed,
      wins: stats.wins,
      winRate: stats.winRate,
      mvpCount: mvpCountMap.get(player.id) ?? 0,
      weeksPlayed: stats.weeksPlayed,
      setsWon: stats.setsWon,
      setDiff: stats.setDiff,
    };
  });

  function top(metric: (r: (typeof rows)[number]) => number, minMatches = 0): LeaderboardEntry[] {
    return rows
      .filter((r) => r.matchesPlayed >= minMatches)
      .map((r) => ({ playerId: r.playerId, name: r.name, value: metric(r) }))
      .sort((a, b) => b.value - a.value)
      .slice(0, LEADERBOARD_SIZE);
  }

  return {
    minMatchesForWinRate: DEFAULT_MIN_MATCHES_FOR_WIN_RATE,
    mostMatches: top((r) => r.matchesPlayed),
    mostWins: top((r) => r.wins),
    highestWinRate: top((r) => r.winRate, DEFAULT_MIN_MATCHES_FOR_WIN_RATE),
    mostMvps: top((r) => r.mvpCount),
    mostWeeksPlayed: top((r) => r.weeksPlayed),
    mostSetsWon: top((r) => r.setsWon),
    bestSetDiff: top((r) => r.setDiff),
  };
}
```

```tsx
import { useState } from 'react';

interface PlayerAvatarProps {
  name: string;
  avatarUrl: string | null;
  sizeClassName?: string;
}

export function PlayerAvatar({ name, avatarUrl, sizeClassName = 'w-10 h-10' }: PlayerAvatarProps) {
  const [imageFailed, setImageFailed] = useState(false);
  const initials = name
    .split(' ')
    .filter(Boolean)
    .slice(0, 2)
    .map((part) => part[0]?.toUpperCase())
    .join('');

  if (avatarUrl && !imageFailed) {
    return (
      // eslint-disable-next-line @next/next/no-img-element -- avatarUrl is an arbitrary admin-entered external domain; next/image would require a remotePatterns allowlist per domain, which isn't practical here.
      <img
        src={avatarUrl}
        alt={name}
        onError={() => setImageFailed(true)}
        className={`${sizeClassName} rounded-full object-cover`}
      />
    );
  }

  return (
    <div
      className={`${sizeClassName} rounded-full bg-slate-200 text-slate-700 flex items-center justify-center text-sm font-medium`}
      aria-label={name}
    >
      {initials || '?'}
    </div>
  );
}
```

```tsx
      {!loading && error && <p className="text-red-600 text-sm">{error}</p>}

      {!loading && !error && overview && (
        <section className="grid grid-cols-2 sm:grid-cols-3 gap-3 mb-6">
          <Card label="Toplam Oyuncu" value={overview.totalPlayers} />
          <Card label="Toplam Hafta" value={overview.totalWeeks} />
          <Card label="Toplam Maç" value={overview.totalMatches} />
          <Card label="Toplam Set" value={overview.totalSets} />
          <Card label="Ortalama Haftalık Katılımcı" value={overview.averageWeeklyAttendance.toFixed(1)} />
          <Card
            label="En Yüksek Katılımlı Hafta"
            value={overview.mostAttendedWeek ? overview.mostAttendedWeek.attendeeCount : '-'}
          />
        </section>
      )}

      {!loading && !error && leaderboards && (
        <section className="space-y-6 mb-6">
          <Leaderboard title="En Çok Maç Oynayan" entries={leaderboards.mostMatches} sortBy="matches" />
          <Leaderboard title="En Çok Galibiyet Alan" entries={leaderboards.mostWins} sortBy="wins" />
          <Leaderboard
            title="En Yüksek Kazanma Oranı"
            entries={leaderboards.highestWinRate}
            sortBy="winRate"
            note={`En az ${leaderboards.minMatchesForWinRate} maç oynamış oyuncular arasında`}
            suffix="%"
          />
          <Leaderboard title="En Çok Haftanın Oyuncusu" entries={leaderboards.mostMvps} sortBy="mvps" />
          <Leaderboard title="En Fazla Hafta Katılımı" entries={leaderboards.mostWeeksPlayed} sortBy="matches" />
          <Leaderboard title="En Fazla Set Kazanan" entries={leaderboards.mostSetsWon} sortBy="matches" />
          <Leaderboard title="En İyi Set Averajı" entries={leaderboards.bestSetDiff} sortBy="matches" />
        </section>
      )}

      {!loading && !error && teamRecords && (
        <section className="space-y-2">
          <h2 className="text-lg font-semibold mb-2">Maç Rekorları</h2>
          <MatchRecordRow label="En Farklı Biten Maç" record={teamRecords.biggestBlowout} />
          <MatchRecordRow label="En Yakın Biten Maç" record={teamRecords.closestMatch} />
          <MatchRecordRow label="En Yüksek Skorlu Maç" record={teamRecords.highestScoringMatch} />
        </section>
      )}
    </main>
  );
}

function Card({ label, value }: { label: string; value: string | number }) {
  return (
    <div className="border rounded p-3">
      <p className="text-xs text-slate-600">{label}</p>
      <p className="text-lg font-bold">{value}</p>
    </div>
  );
}

function Leaderboard({
  title,
  entries,
  sortBy,
  note,
  suffix = '',
}: {
  title: string;
  entries: LeaderboardEntry[];
  sortBy: string;
  note?: string;
  suffix?: string;
}) {
  return (
    <div>
      <div className="flex justify-between items-baseline">
        <h2 className="text-lg font-semibold">{title}</h2>
        <Link href={`/players?sortBy=${sortBy}&order=desc`} className="text-sm underline">
          Tümünü Gör
        </Link>
      </div>
      {note && <p className="text-xs text-slate-600 mb-1">{note}</p>}
      {entries.length === 0 ? (
        <p className="text-slate-600 text-sm">Henüz veri yok.</p>
      ) : (
        <ol className="text-sm space-y-1">
          {entries.slice(0, 5).map((e, i) => (
            <li key={e.playerId} className="flex justify-between">
              <span>
                {MEDALS[i] ?? `${i + 1}.`} {e.name}
              </span>
              <span>
                {e.value.toFixed(0)}
                {suffix}
              </span>
            </li>
          ))}
        </ol>
      )}
    </div>
  );
}

function MatchRecordRow({ label, record }: { label: string; record: MatchRecordSummary | null }) {
  if (!record) {
    return <p className="text-sm text-slate-600">{label}: henüz veri yok.</p>;
  }
  return (
    <p className="text-sm">
      <span className="font-medium">{label}:</span> {record.teamAName} - {record.teamBName} (
      {new Date(record.weekDate).toLocaleDateString('tr-TR')})
    </p>
  );
}
```
